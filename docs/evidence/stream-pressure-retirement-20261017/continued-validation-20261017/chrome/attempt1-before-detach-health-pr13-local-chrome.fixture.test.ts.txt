function deferred<T>() { let resolve!: (value: T) => void; const promise = new Promise<T>(done => { resolve = done; }); return { promise, resolve }; }
async function until(check: () => boolean | Promise<boolean>, label: string, ms = 7000) {
  const end = performance.now() + ms;
  while (!await check()) { if (performance.now() > end) throw new Error(`Fixture deadline: ${label}`); await Bun.sleep(10); }
}
function alive(pid: number) { try { process.kill(pid, 0); return true; } catch (error: any) { if (error.code === "ESRCH") return false; throw error; } }
function frames(wire: string): any[] { return wire.split("\n").filter(line => line.startsWith("data: {")).map(line => JSON.parse(line.slice(6))); }
function responseEvidence(label: string, httpStatus: number, wire: string) {
  const streaming = wire.startsWith("data:") || wire.startsWith("event:");
  const ends = streaming ? frames(wire).filter(event => ["response.completed", "response.failed", "response.incomplete"].includes(event.type)).map(event => event.response) : [JSON.parse(wire)];
  return { label, httpStatus, streaming, terminalCount: streaming ? ends.length : ends.filter(value => ["completed", "failed", "incomplete"].includes(value.status)).length,
    doneCount: streaming ? wire.split("data: [DONE]").length - 1 : null,
    terminals: ends.map(value => {
      const text = (value.output ?? []).flatMap((item: any) => item.content ?? []).filter((part: any) => part.type === "output_text").map((part: any) => part.text ?? "").join("");
      return { status: value.status ?? null, code: value.error?.code ?? null, retryable: value.retryable ?? null,
        textLength: text.length, textBytes: Buffer.byteLength(text), textSha256: createHash("sha256").update(text).digest("hex") };
    }) };
}
function terminal(wire: string) {
  const ends = frames(wire).filter(event => ["response.completed", "response.failed", "response.incomplete"].includes(event.type));
  expect(ends).toHaveLength(1); expect(wire.split("data: [DONE]")).toHaveLength(2); return ends[0].response;
}

localTest("Local Chrome: shutdown retains its listener until its owned Chrome child exits", async () => {
  const f = await fixture("shutdown"); let reader: ReadableStreamDefaultReader<Uint8Array> | undefined;
  try {
    reader = await f.prefix(); const old = await f.owner(); f.signalShutdown();
    await old.closing.promise; await reader.cancel(); reader.releaseLock(); reader = undefined;
    expect(old.signal?.aborted).toBe(true); expect(alive(old.child!.pid!)).toBe(true);
    expect(await f.health("shutdown-waits-real-child")).toMatchObject({ accepting_turns: false, active_browser_turns: 1 });
    f.note("shutdown-physical-retirement-pending", { pid: old.child!.pid, alive: alive(old.child!.pid!), exitObserved: old.exitObserved });
    const refused = await f.control("/admin/shutdown"); expect(refused.status).toBe(409);
    const refusedBody = await refused.json();
    f.note("shutdown-refused-while-child-alive", { httpStatus: refused.status, result: refusedBody });
    expect(refusedBody).toMatchObject({ status: "refused", active_browser_turns: 1 });
    old.closeGate.resolve(); await old.done.promise;
    expect(old.exitObserved).toBe(true); expect(alive(old.child!.pid!)).toBe(false);
    await until(async () => { try { await f.health("shutdown-close-poll", false); return false; } catch { return true; } }, "shutdown listener closed");
    f.note("shutdown-listener-unreachable-after-child-close", { pid: old.child!.pid, alive: alive(old.child!.pid!), exitObserved: old.exitObserved });
  } finally { await reader?.cancel().catch(() => {}); await f.close(); }
}, 30_000);
